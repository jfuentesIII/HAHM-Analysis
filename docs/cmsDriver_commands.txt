# --- Run3 Step 1: GEN-SIM (GENSIM) ---
# Fragment: Configuration/GenFragments/python/HAHM_frag_file_LHEParams_cff.py
#   - Uses MCTunesRun3ECM13p6TeV.PythiaCP5Settings_cfi
#   - comEnergy = 13600.0
#   - BeamRemnants:reconnectRange line should be COMMENTED OUT
#
cmsDriver.py Configuration/GenFragments/python/HAHM_frag_file_LHEParams_cff.py \
--filein file:dummy.lhe \
--fileout file:GEN-SIM.root \
--customise Configuration/DataProcessing/Utils.addMonitoring \
--mc --eventcontent RAWSIM \
--datatier GEN-SIM \
--conditions 140X_mcRun3_2024_realistic_v26 \
--beamspot DBrealistic \
--step GEN,SIM \
--geometry DB:Extended \
--era Run3_2024 \
--python_filename tmp_GEN-SIM.py \
--no_exec -n -1 \
--nThreads 4
#
# Manual edits after generation:
#   1. Change PoolSource to LHESource (if needed)
#   2. Replace the dummy.lhe filename with PLACEHOLDER_LHE:
#      fileNames = cms.untracked.vstring('file:PLACEHOLDER_LHE')


# --- Run3 Step 2: DIGI-HLT (DIGIHLT) ---
# Uses premixing workflow with PREMIX pileup files.
#
cmsDriver.py \
--fileout file:DIGI-L1-DIGI2RAW-HLT.root \
--mc --eventcontent PREMIXRAW \
--customise Configuration/DataProcessing/Utils.addMonitoring \
--datatier GEN-SIM-RAW \
--conditions 140X_mcRun3_2024_realistic_v26 \
--step DIGI,DATAMIX,L1,DIGI2RAW,HLT:2024v14 \
--procModifiers premix_stage2 \
--datamix PreMix \
--geometry DB:Extended \
--era Run3_2024 \
--pileup_input das:/Neutrino_E-10_gun/RunIIISummer24PrePremix-Premixlib2024_140X_mcRun3_2024_realistic_v26-v1/PREMIX \
--pileup_dasoption "--limit 50" \
--python_filename tmp_DIGI-L1-DIGI2RAW-HLT.py \
--no_exec -n -1 \
--nThreads 8
#
# Manual edits after generation:
#   None needed. Use as-is.
#
# Notes:
#   - --datamix PreMix and --procModifiers premix_stage2 are required because
#     the pileup dataset is PREMIX tier (pre-digitized pileup).
#   - --pileup_dasoption "--limit 50" limits pileup files to 50 to avoid
#     excessive memory usage (~500k files in the full dataset).
#   - Source: https://twiki.cern.ch/twiki/bin/view/Main/Bsm3gRun3PrivateMC
